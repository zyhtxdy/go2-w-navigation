cmake_minimum_required(VERSION 3.16)
project(go2_offline_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(terrain_costmap REQUIRED)
add_executable(terrain_map_builder src/terrain_map_builder.cpp)
target_include_directories(terrain_map_builder SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(terrain_map_builder ${PCL_LIBRARIES})
ament_target_dependencies(terrain_map_builder rclcpp terrain_costmap)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS terrain_map_builder DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/build_terrain_map scripts/offline_map_server scripts/localization_adapter scripts/validate_replay
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
# Keep configuration canonical; the offline launcher rewrites only its own copy.
install(FILES ../nav2/navigation_bringup/config/nav2_mppi_params.yaml
  DESTINATION share/${PROJECT_NAME}/config)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_obstacle_health test/test_obstacle_health.py TIMEOUT 30)
  ament_add_pytest_test(test_offline_navigation test/test_offline_navigation.py TIMEOUT 120)
  ament_add_pytest_test(test_adapter_edges test/test_adapter_edges.py TIMEOUT 45)
  ament_add_pytest_test(test_navigation_guard test/test_navigation_guard.py TIMEOUT 30)
  ament_add_pytest_test(test_execution_configuration test/test_execution_configuration.py TIMEOUT 30)
  ament_add_pytest_test(test_map_identity test/test_map_identity.py TIMEOUT 30)
  ament_add_pytest_test(test_planning_settings test/test_planning_settings.py TIMEOUT 30)
  ament_add_pytest_test(test_planning_evaluation test/test_planning_evaluation.py TIMEOUT 45)
  ament_add_pytest_test(test_obstacle_evidence test/test_obstacle_evidence.py TIMEOUT 45)
  ament_add_pytest_test(test_replay_log_preservation test/test_replay_log_preservation.py TIMEOUT 30)
  ament_add_pytest_test(test_near_obstacle_planning test/test_near_obstacle_planning.py TIMEOUT 60)
endif()
ament_package()
